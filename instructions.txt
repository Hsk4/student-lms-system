--muted-foreground: #6E5A60  (captions, labels)

--primary-hover: #6A1830     (hover/pressed state)
--accent: #F1E3E6            (selected rows, active nav bg, chips)

dark warm background (jese #1C1214), dark card (#2A1A20), proper contrast
